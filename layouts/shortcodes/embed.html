{{- $url := .Get "url" -}} {{- with .Get "ref" -}} {{- $url = ref $.Page . -}} {{- end -}} {{- $fallbackTitle := .Get "title" | default $url -}} {{- $parsedURL := urls.Parse $url -}} {{- $result := try (resources.GetRemote $url (dict "timeout" "3s")) -}} {{- if $result.Err -}} {{ $fallbackTitle }} {{- else -}} {{- with $result.Value -}} {{- /* Go regular expressions match invalid UTF-8 bytes as U+FFFD. */ -}} {{- if findRE `\x{FFFD}` .Content -}} {{ $fallbackTitle }} {{- else -}} {{- $metadata := partial "embed/metadata.html" .Content -}} {{- if not $metadata.title -}} {{- $metadata = merge $metadata (dict "title" $fallbackTitle) -}} {{- end -}} {{- partial "embed/card.html" (dict "url" $url "host" $parsedURL.Host "scheme" $parsedURL.Scheme "metadata" $metadata) -}} {{- end -}} {{- else -}} {{ $fallbackTitle }} {{- end -}} {{- end -}}
